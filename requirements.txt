fastapi                      # the web framework
uvicorn                      # the server that runs FastAPI
python-multipart             # needed for file uploads (paper PDFs)
pymupdf                      # PDF parsing. AGPL-3.0 — a recorded project decision.
                             # Chosen over text-only extractors because span-level
                             # bbox/colour/size are required for reading order,
                             # invisible-text stripping and heading detection.
pydantic                     # data validation
pydantic-settings            # reads config from environment variables
sqlalchemy                   # ORM — talks to Postgres
asyncpg                      # Postgres driver, async
psycopg[binary]              # Postgres driver, sync — Alembic against local Postgres
pg8000                       # Postgres driver, sync, pure Python — Alembic against
                             # Cloud SQL. The connector's psycopg path needs a unix
                             # socket, which Windows does not have.
alembic                      # database migrations
pgvector                     # Python bindings for the vector column type
google-cloud-storage         # uploads PDFs to Cloud Storage
google-cloud-tasks           # queues ingestion jobs
google-auth                  # verifies the OIDC token on Cloud Tasks pushes
firebase-admin               # verifies login tokens
cloud-sql-python-connector   # secure connection from Cloud Run to Cloud SQL
google-adk                   # the agent framework
google-genai                 # talks to Gemini
httpx                        # HTTP client (internal calls, testing)
tenacity                     # retry logic for flaky external calls
